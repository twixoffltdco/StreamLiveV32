# Красивые ссылки без .php — добавь это в свой основной .htaccess в корне проекта
# (не затирай существующие правила, просто вставь блок ниже).
#
# ВАЖНО: mod_rewrite должен быть включён на хостинге — на InfinityFree и большинстве
# обычных Apache-хостингов он включён по умолчанию, но если правила не сработают,
# первым делом проверь именно это в панели хостинга.

<IfModule mod_rewrite.c>
  RewriteEngine On

  # /video/nazvanie-slug  →  video.php?slug=nazvanie-slug
  RewriteRule ^video/([a-z0-9-]+)/?$ video.php?slug=$1 [L,QSA]

  # /channel/nazvanie-slug  →  channel.php?slug=nazvanie-slug
  RewriteRule ^channel/([a-z0-9-]+)/?$ channel.php?slug=$1 [L,QSA]

  # /s/nazvanie-slug  →  s.php?slug=nazvanie-slug (задеплоенные GitHub-сервисы)
  RewriteRule ^s/([a-z0-9-]+)/?$ s.php?slug=$1 [L,QSA]

  # /u/username  →  profile.php?username=username
  RewriteRule ^u/([a-zA-Z0-9_-]+)/?$ profile.php?username=$1 [L,QSA]
</IfModule>
